exports lower-level primitives; it does not expose the variants documented here.

## Usage

Import the stylesheet and mount `ExegiaProvider` once at your application root.
The provider supplies tooltip and portal infrastructure. This complete example
includes that setup; if your app already has it, use just `PageBreadcrumb`.

```tsx
"use client"

import { Breadcrumb, ExegiaProvider } from "@exegia/corpora-ui"
import "@exegia/corpora-ui/index.css"

export function PageBreadcrumb() {
  return (
    <Breadcrumb.Root aria-label="Page location">
      <Breadcrumb.List>
        <Breadcrumb.Item
          variant="link"
          label="Home"
          href="/"
          tooltip="Return to the homepage"
        />
        <Breadcrumb.Separator />
        <Breadcrumb.Item variant="default" label="Library">
          <Breadcrumb.Page>Library</Breadcrumb.Page>
        </Breadcrumb.Item>
      </Breadcrumb.List>
    </Breadcrumb.Root>
  )
}

export default function App() {
  return (
    <ExegiaProvider>
      <PageBreadcrumb />
    </ExegiaProvider>
  )
}
```

The following examples each define every component they use. Render them inside
your existing `ExegiaProvider`; no story imports or documentation-only helpers
are needed.

## TOC and menu items

Supply components with their data already configured. A `toc` component renders
content inside the breadcrumb's popover. A `menu` component owns its popup and
must use the supplied `children` element as its trigger. Define these components
outside the breadcrumb render function to keep their identity stable.

```tsx
"use client"

import { useId, type ReactElement } from "react"
import { Breadcrumb, MenuCommand, TOC } from "@exegia/corpora-ui"

function ChapterContents() {
  return (
    <div className="w-64 max-w-full">
      <TOC.Root
        items={[
          {
            id: "genesis-1",
            label: "Genesis 1",
            link: "#genesis-1-chapter",
            level: 1,
            type: "chapter",
          },
          {
            id: "genesis-2",
            label: "Genesis 2",
            link: "#genesis-2-chapter",
            level: 1,
            type: "chapter",
          },
        ]}
      />
    </div>
  )
}

function BookMenu({ children }: { children: ReactElement }) {
  return (
    <MenuCommand
      side="bottom"
      align="start"
      items={[
        { id: "genesis-1", label: "Genesis 1", href: "#genesis-1-chapter" },
        { id: "genesis-2", label: "Genesis 2", href: "#genesis-2-chapter" },
      ]}
    >
      {children}
    </MenuCommand>
  )
}

export function ReaderLocation() {
  const id = useId()

  return (
    <>
      <Breadcrumb.Root aria-label="Reading location">
        <Breadcrumb.List>
          <Breadcrumb.Item variant="link" label="Home" href="/" />
          <Breadcrumb.Separator />
          <Breadcrumb.Item
            variant="menu"
            id={`${id}-book-menu`}
            label="Genesis"
            Component={BookMenu}
            tooltip="Choose a chapter"
          />
          <Breadcrumb.Separator />
          <Breadcrumb.Item
            variant="toc"
            id={`${id}-chapter-toc`}
            label="Contents"
            Component={ChapterContents}
            tooltip="Browse the table of contents"
          />
          <Breadcrumb.Separator />
          <Breadcrumb.Item variant="default" label="Chapter 1">
            <Breadcrumb.Page>Chapter 1</Breadcrumb.Page>
          </Breadcrumb.Item>
        </Breadcrumb.List>
      </Breadcrumb.Root>
      <section id="genesis-1-chapter">
        <h2>Genesis 1</h2>
        <p>Chapter one content.</p>
      </section>
      <section id="genesis-2-chapter">
        <h2>Genesis 2</h2>
        <p>Chapter two content.</p>
      </section>
    </>
  )
}
```

The item `id` identifies the trigger button, not the selected TOC node or menu
option. Selection, navigation, popup settings, and data belong to the supplied
component. The breadcrumb does not pass its `id` or `label` to that component.

## Separator symbols and menus

Use `symbol="chevron"` (the default), `symbol="slash"`, or custom text/icons as
`children`. A menu separator uses the same preconfigured-menu contract as a menu
item, but renders an icon-sized button instead of a text-labelled item.

```tsx
"use client"

import type { ReactElement } from "react"
import { Breadcrumb, MenuCommand } from "@exegia/corpora-ui"

function LocationMenu({ children }: { children: ReactElement }) {
  return (
    <MenuCommand
      side="bottom"
      align="start"
      items={[
        { id: "home", label: "Home", href: "/" },
        { id: "library", label: "Library", href: "/library" },
      ]}
    >
      {children}
    </MenuCommand>
  )
}

export function SeparatorExamples() {
  const examples = [
    { label: "Chevron", separator: <Breadcrumb.Separator /> },
    { label: "Slash", separator: <Breadcrumb.Separator symbol="slash" /> },
    {
      label: "Custom symbol",
      separator: <Breadcrumb.Separator>·</Breadcrumb.Separator>,
    },
    {
      label: "Menu button",
      separator: (
        <Breadcrumb.Separator
          variant="menu"
          label="Choose a location"
          Component={LocationMenu}
        />
      ),
    },
  ]

  return (
    <div className="gap-4 lg:grid-cols-2 grid">
      {examples.map(({ label, separator }) => (
        <Breadcrumb.Root key={label} aria-label={`${label} example`}>
          <Breadcrumb.List>
            <Breadcrumb.Item variant="link" label="Home" href="/" />
            {separator}
            <Breadcrumb.Item variant="default" label="Library">
              <Breadcrumb.Page>Library</Breadcrumb.Page>
            </Breadcrumb.Item>
          </Breadcrumb.List>
        </Breadcrumb.Root>
      ))}
    </div>
  )
}
```

## API notes

<span id="breadcrumb-variants-section" />

### Item variants

`Breadcrumb.Item` is a discriminated union. `variant` and `label` are required
for every item; only the props belonging to the selected variant are accepted.

| Variant     | Additional required props | Rendered behavior                                                                                                         |
| ----------- | ------------------------- | ------------------------------------------------------------------------------------------------------------------------- |
| `"default"` | `children`                | Renders the supplied content inside a disabled, small ghost button. Use for noninteractive content; it does not navigate. |
| `"link"`    | `href`                    | Renders a small ghost-styled anchor with normal browser navigation.                                                       |
| `"toc"`     | `id`, `Component`         | Renders a labelled button that opens a start-aligned popover containing `Component`.                                      |
| `"menu"`    | `id`, `Component`         | Passes a labelled button to `Component` as its trigger; the supplied component owns the menu.                             |

### Shared item props

| Prop        | Type                                     | Required / default   | Impact                                                                                                                                                                                                          |
| ----------- | ---------------------------------------- | -------------------- | --------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------- |
| `variant`   | `"default" \| "link" \| "toc" \| "menu"` | Required; no default | Selects the rendered control and the required variant-specific props.                                                                                                                                           |
| `label`     | `string`                                 | Required             | Visible text for link, TOC, and menu buttons. For `default`, the visible content comes from `children` instead. Also names the TOC popup and is the shared tooltip's fallback text.                             |
| `tooltip`   | `ReactNode`                              | Optional; off        | Hint shown by the shared tooltip. `null`, `undefined`, and `false` disable it; `label` alone does not enable it. The disabled `default` button is not keyboard-focusable, so use hints on interactive variants. |
| `className` | `string`                                 | Optional             | Styles the outer item wrapper, not the internal button, anchor, or popup.                                                                                                                                       |

### Variant-specific item props

| Prop        | Available on   | Type                                        | Required | Impact                                                                                                                                                                    |
| ----------- | -------------- | ------------------------------------------- | -------- | ------------------------------------------------------------------------------------------------------------------------------------------------------------------------- |
| `children`  | `default`      | `ReactNode`                                 | Yes      | Content inside the disabled button. Text, icons, and `Breadcrumb.Page` are supported; do not nest interactive controls.                                                   |
| `href`      | `link`         | `string`                                    | Yes      | Destination URL or fragment.                                                                                                                                              |
| `onClick`   | `link`, `menu` | `() => void`                                | No       | Runs when the link or menu trigger is clicked. It is not a menu-option selection callback and receives no event argument. On links it does not replace normal navigation. |
| `id`        | `toc`, `menu`  | `string`                                    | Yes      | DOM ID of the trigger button. Keep it unique, for example with React's `useId()`.                                                                                         |
| `Component` | `toc`          | `ComponentType`                             | Yes      | Receives no props. Supplies its own TOC data and content; the breadcrumb owns the popover.                                                                                |
| `Component` | `menu`         | `ComponentType<{ children: ReactElement }>` | Yes      | Receives the breadcrumb trigger as `children`. Supplies menu data and forwards the trigger to the menu primitive.                                                         |

`Breadcrumb.Item` does **not** accept arbitrary DOM props, `overlay`, `items`,
`open`, `onOpenChange`, or `portalProps`. Configure overlay behavior inside your
menu component. The built-in TOC popover does not expose these settings through
the item API.

### Separator props

| Prop             | Type                                                       | Required / default                             | Impact                                                                                                                               |
| ---------------- | ---------------------------------------------------------- | ---------------------------------------------- | ------------------------------------------------------------------------------------------------------------------------------------ |
| `variant`        | `"default" \| "menu"`                                      | Optional; `"default"` behavior                 | Decorative symbol, or a button that opens a supplied menu.                                                                           |
| `symbol`         | `"chevron" \| "slash"`                                     | Optional; `"chevron"`                          | Selects the built-in symbol in either variant.                                                                                       |
| `children`       | `ReactNode`                                                | Optional                                       | Custom text or icon. Non-nullish children override `symbol`; `false` renders no symbol. Keep the content noninteractive.             |
| `className`      | `string`                                                   | Optional                                       | Styles the outer `li`, not the internal menu button or popup.                                                                        |
| `label`          | `string`                                                   | Required for `menu`; unavailable for `default` | Accessible name for the icon-only menu button. Not displayed as visible text.                                                        |
| `Component`      | `ComponentType<{ children: ReactElement }>`                | Required for `menu`; unavailable for `default` | Preconfigured menu receiving the separator button as its trigger.                                                                    |
| `disabled`       | `boolean`                                                  | Optional for `menu`; false behavior            | Disables the menu trigger. Unavailable for decorative separators.                                                                    |
| Other `li` props | `ComponentProps<"li">`, excluding `role` and `aria-hidden` | Optional                                       | Forwarded to the outer list item. For example, `id` targets the wrapper and event handlers attach to it, not directly to the button. |

Decorative separators are hidden from assistive technology. A menu separator
keeps its button accessible and hides only the decorative symbol. `role` and
`aria-hidden` are managed internally, so they cannot be supplied as props.

### Composition components

These components accept the native props listed below, including `children`,
`className`, `style`, and applicable event/ARIA attributes, unless noted.

| Component             | Props                                   | Defaults and impact                                                                                                                                                                     |
| --------------------- | --------------------------------------- | --------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------- |
| `Breadcrumb.Root`     | `ComponentProps<"nav">`                 | Renders navigation with `aria-label="breadcrumb"` and mounts the shared tooltip popup. Override the label to distinguish multiple trails. Keep items inside a root.                     |
| `Breadcrumb.List`     | `ComponentProps<"ol">`                  | Renders the wrapping, flex-layout ordered list. Additional classes apply to the list. Place separators explicitly between items.                                                        |
| `Breadcrumb.Page`     | `ComponentProps<"span">`                | Adds `aria-current="page"` and current-location styling. Put it inside a `default` item for the active page. Native props can override the default ARIA attribute.                      |
| `Breadcrumb.Link`     | Base UI `useRender.ComponentProps<"a">` | Low-level anchor with hover styling. Supports native anchor props plus `render` to replace the element. Use `Item variant="link"` for the composed item API.                            |
| `Breadcrumb.Ellipsis` | `ComponentProps<"span">`                | Static more-horizontal icon, hidden from assistive technology by default. It does not collapse items or open a menu. Its icon content is fixed rather than supplied through `children`. |

### Type exports

| Export from `@exegia/corpora-ui`       | Purpose                                                                                                                               |
| -------------------------------------- | ------------------------------------------------------------------------------------------------------------------------------------- |
| `IBreadcrumbItemProps<Variant, Label>` | Item discriminated union. `Variant` defaults to all item variants; `Label` defaults to `string` and can constrain labels to literals. |
| `TBreadcrumbVariant`                   | Union of `"default"`, `"link"`, `"toc"`, and `"menu"`.                                                                                |
| `IBreadcrumbSeparatorProps`            | Decorative/menu separator union, including wrapper props and symbol selection.                                                        |
| `TBreadcrumbSeparatorSymbol`           | Union of `"chevron"` and `"slash"`.                                                                                                   |
